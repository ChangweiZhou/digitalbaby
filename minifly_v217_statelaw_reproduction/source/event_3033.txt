"""MiniWorld v1: people move, pick up and drop objects. Questions about people and objects. One life, fixed learner.
Learner (no positional roles, no supplied operations per verb):
  store: every event -> bag of its words (key = byte n-grams of the event) PLUS an 'enrichment' bag: for each word of the
         event, the words recalled for that word at event time (generic "what did I know about each participant then").
  answer: candidates tagged d (words of the recalled event), e (its enrichment, i.e. event-time knowledge),
          h (words recalled now via each word of the recalled event: second hop).
  choice: naive-Bayes evidence from feedback over features: identity, slot context, and (tag x word of the recalled event)
          and (tag x via-word slot context). Random tie-breaking."""
import numpy as np, random, os, json
E=os.environ.get
SYL=['ka','lo','mi','ru','te','sa','no','vi','da','pe','zu','ho','bi','fe','go','ra','li','mu','to','ne']
MOVE_TR=['{N} went to the {L}.','{N} moved to the {L}.','{N} is now in the {L}.','The {L} is where {N} went.','{N} ran into the {L}.','Now {N} is in the {L}.','{N} walked over to the {L}.','{N} arrived at the {L}.']
MOVE_TE=['{N} strolled into the {L}.','At the {L} one finds {N}.','{N} has reached the {L}.']
GET=['{N} got the {O}.','{N} picked up the {O}.','{N} took the {O}.','{N} grabbed the {O}.']
DROP=['{N} dropped the {O}.','{N} put down the {O}.','{N} left the {O}.','{N} discarded the {O}.']
def word(r): return ''.join(r.choice(SYL) for _ in range(r.choice([2,3])))
def world(r,pool):
    return (list({word(r).capitalize() for _ in range(6)})[:4], list({word(r) for _ in range(9)})[:6], list({word(r) for _ in range(5)})[:3], r.sample(pool,2))
def stream(r,w,nq):
    names,places,objs,tpl=w; loc={n:r.choice(places) for n in names}; hold={}; oloc={}; out=[(  's',r.choice(tpl).format(N=n,L=loc[n])) for n in names]; q=0
    while q<nq:
        n=r.choice(names); x=r.random(); held=[o for o in objs if hold.get(o)==n]; free=[o for o in objs if o not in hold]
        if x<0.5:
            l=r.choice([p for p in places if p!=loc[n]]); loc[n]=l; out.append(('s',r.choice(tpl).format(N=n,L=l)))
        elif x<0.75 and free:
            o=r.choice(free); hold[o]=n; oloc.pop(o,None); out.append(('s',r.choice(GET).format(N=n,O=o)))
        elif held:
            o=r.choice(held); del hold[o]; oloc[o]=loc[n]; out.append(('s',r.choice(DROP).format(N=n,O=o)))
        else: continue
        if r.random()<0.4:
            known=[o for o in objs if o in hold or o in oloc]
            if known and r.random()<0.6:
                o=r.choice(known); out.append(('q',(f'Where is the {o}?',loc[hold[o]] if o in hold else oloc[o],'obj'))); q+=1
            else:
                m=r.choice(names); out.append(('q',(f'Where is {m}?',loc[m],'per'))); q+=1
    return out
class Learner:
    N=1<<15; V=2048
    def __init__(s,seed):
        s.M=np.zeros((s.N,2*s.V),np.float32); s.lex={}; s.words=[]; s.CF={}; s.EV={}; s.EC={}; s.DF={}; s.NQ=0; s.rng=random.Random(seed)
        s.HIT={}; s.TRY={}; s.HT=[0,0]
    def wid(s,w):
        if w not in s.lex: s.lex[w]=len(s.words); s.words.append(w)
        return s.lex[w]
    def units(s,h,flat=False):
        d={}
        for g in [h[i:i+k] for k in (2,3,4) for i in range(len(h)-k+1)]:
            wt=1.0 if flat else np.log((1+s.NQ)/(1+s.DF.get(g,0)))
            for j in range(4): d[(hash(g)*2654435761+j*40503)%s.N]=wt
        return np.fromiter(d,np.int64),np.fromiter(d.values(),float)
    def recall(s,cue):
        ix,v=s.units(cue.encode()); r=v@s.M[ix]; return r
    def see(s,x):
        ws=x.rstrip('.').split(); b=np.zeros(2*s.V,np.float32)
        for w in ws: b[s.wid(w)]=1/len(ws)
        for w in ws:                                     # enrichment: what each participant recalls right now (top word)
            r=s.recall(w)[:s.V]; r[[s.lex[u] for u in ws]]=0
            if r.max()>0: b[s.V+int(np.argmax(r))]+=1/len(ws)
        ix,_=s.units(x.encode(),True); s.M[ix]=b
        for i,w in enumerate(ws): s.CF[w]=[('p',ws[i-1].lower() if i else '^'),('n',ws[i+1].lower() if i+1<len(ws) else '$')]
        s.EVLAST=ws
    def feats(s,c,tag,ev,via=None):
        w=s.words[c]; f=[('id',w)]+[('ctx',)+x for x in s.CF.get(w,[])]+[(tag,'ev',u.lower()) for u in ev if not u[0].isupper()] + [(tag,)]
        if via: f+=[(tag,'via')+x for x in s.CF.get(via,[])]
        return f
    def ask(s,q):
        qb=q.rstrip('?'); r=s.recall(qb); cands={}
        d=r[:s.V]; ev=[s.words[i] for i in np.nonzero(d>0)[0]]
        evtop=[s.words[i] for i in np.argsort(-d)[:8] if d[i]>0.5*d.max()] if d.max()>0 else []
        for c in np.nonzero(d>0)[0]: cands[(c,'d',None)]=d[c]
        e=r[s.V:]
        for c in np.nonzero(e>0)[0]: cands[(c,'e',None)]=e[c]
        for vw in evtop:
            r2=s.recall(vw)[:s.V]
            if r2.max()>0:
                for c in np.nonzero(r2>0.5*r2.max())[0]: cands[(c,'h',vw)]=r2[c]/r2.max()
        if not cands: return None,[],evtop
        p0=(s.HT[0]+1)/(s.HT[1]+2); lo=lambda x: np.log(x/(1-x))
        def score(k):
            c,tag,via=k; fs=s.feats(c,tag,evtop,via)
            return lo(p0)+sum(lo((s.HIT.get(f,0)+2*p0)/(s.TRY.get(f,0)+2))-lo(p0) for f in fs)
        sc={k:score(k) for k in cands}; best=max(sc.values()); ties=[k for k in sc if sc[k]>=best-1e-9]
        k=s.rng.choice(ties); return s.words[k[0]],list(cands),evtop
    def feedback(s,q,cands,evtop,a):
        if a is not None:
            for c,tag,via in cands:
                ok=s.words[c]==a; s.HT[0]+=ok; s.HT[1]+=1
                for f in s.feats(c,tag,evtop,via): s.TRY[f]=s.TRY.get(f,0)+1; s.HIT[f]=s.HIT.get(f,0)+ok
        qb=q.rstrip('?').encode()
        for g in {qb[i:i+k] for k in (2,3,4) for i in range(len(qb)-k+1)}: s.DF[g]=s.DF.get(g,0)+1
        s.NQ+=1
def live(L,r,w,nq,fb,rec=None):
    for k,x in stream(r,w,nq):
        if k=='s': L.see(x)
        else:
            q,a,kind=x; out,c,ev=L.ask(q)
            if rec is not None: rec.append((out==a,kind))
            L.feedback(q,c,ev,{'true':a,'type':r.choice(w[1]),'none':None}[fb])
K=int(E('K','30')); NQ=int(E('NQ','60')); NT=int(E('NT','60')); NS=int(E('NS','8'))
for fb in E('FBS','true,type,none').split(','):
    A=[];B=[]
    for sd in range(NS):
        r=random.Random(sd); La=Learner(sd)
        for _ in range(K): live(La,r,world(r,MOVE_TR),NQ,fb)
        tw=world(random.Random(10_000+sd),MOVE_TE)
        ra=[]; live(La,random.Random(20_000+sd),tw,NT,fb,ra); rb=[]; live(Learner(sd),random.Random(20_000+sd),tw,NT,fb,rb)
        A.append(ra[:NT]); B.append(rb[:NT])
    def acc(X,kind,lo=0,hi=None): v=[o for row in X for o,k in row[lo:hi] if k==kind]; return round(float(np.mean(v)),3) if v else None
    d=[sum(o for o,_ in a)-sum(o for o,_ in b) for a,b in zip(A,B)]
    print(fb,json.dumps(dict(exp_person=acc(A,'per'),exp_object=acc(A,'obj'),fresh_person=acc(B,'per'),fresh_object=acc(B,'obj'),
        exp_first20_obj=acc(A,'obj',0,20),fresh_first20_obj=acc(B,'obj',0,20),transfer_area=round(float(np.mean(d)),2),ci95=round(1.96*float(np.std(d,ddof=1))/np.sqrt(len(d)),2))),flush=True)
